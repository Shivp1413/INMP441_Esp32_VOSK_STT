
#include <WiFi.h>
#include <WiFiUdp.h>
#include "driver/i2s.h"

// INMP441 pins
#define I2S_BCLK_PIN  5
#define I2S_WS_PIN    6
#define I2S_DATA_PIN  4

// Wi-Fi access point
const char* AP_SSID = "mystream";
const char* AP_PASS = "12345678";

// UDP settings
const IPAddress broadcastIP(192, 168, 4, 255);
const uint16_t UDP_PORT = 3333;

// Audio settings
#define SAMPLE_RATE 16000
#define SAMPLES     320
#define GAIN        3

WiFiUDP udp;
int32_t raw[SAMPLES];
int16_t pcm[SAMPLES];

void setup() {
  Serial.begin(115200);

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);

  Serial.print("AP IP: ");
  Serial.println(WiFi.softAPIP());

  i2s_config_t cfg = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = SAMPLE_RATE,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,
    .dma_buf_len = 256,
    .use_apll = false,
    .tx_desc_auto_clear = false,
    .fixed_mclk = 0
  };

  i2s_pin_config_t pins = {
    .bck_io_num = I2S_BCLK_PIN,
    .ws_io_num = I2S_WS_PIN,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_DATA_PIN
  };

  i2s_driver_install(I2S_NUM_0, &cfg, 0, NULL);
  i2s_set_pin(I2S_NUM_0, &pins);
  i2s_zero_dma_buffer(I2S_NUM_0);

  udp.begin(UDP_PORT);

  Serial.println("Streaming louder audio over UDP");
}

void loop() {
  size_t bytesRead = 0;

  esp_err_t err = i2s_read(
    I2S_NUM_0,
    raw,
    sizeof(raw),
    &bytesRead,
    portMAX_DELAY
  );

  if (err != ESP_OK || bytesRead == 0) return;

  int count = bytesRead / sizeof(int32_t);

  for (int i = 0; i < count; i++) {
    // Convert microphone audio to 16-bit PCM
    int32_t sample = raw[i] >> 16;

    // Increase volume
    sample *= GAIN;

    // Prevent overflow and distortion from integer wraparound
    if (sample > 32767) sample = 32767;
    if (sample < -32768) sample = -32768;

    pcm[i] = (int16_t)sample;
  }

  udp.beginPacket(broadcastIP, UDP_PORT);
  udp.write((uint8_t*)pcm, count * sizeof(int16_t));
  udp.endPacket();
}
